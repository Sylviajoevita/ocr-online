@app.route('/')
def home():
    return render_template_string(HTML_PAGE)

@app.route('/ocr', methods=['POST'])
def ocr():
    files = request.files.getlist('images')
    results = []
    for f in files:
        try:
            img = Image.open(f.stream)
            text = pytesseract.image_to_string(img)
            safe_name = "".join(c if c.isalnum() else "_" for c in os.path.splitext(f.filename)[0])[:31] or "Sheet"
            results.append({
                "sheetName": safe_name,
                "fileName": f.filename,
                "text": text
            })
        except Exception as e:
            results.append({
                "sheetName": "Error",
                "fileName": f.filename,
                "text": str(e)
            })
    return jsonify(results)
